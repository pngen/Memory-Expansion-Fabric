
# Helper: add a test executable linked to mef_core with warnings + asan as needed.
function(mef_add_test src)
  add_executable(mef_${src} ${src}.cpp)
  target_link_libraries(mef_${src} PRIVATE mef_core)
  mef_apply_warnings(mef_${src})
  if(MSVC AND MEF_ENABLE_ASAN)
    target_compile_options(mef_${src} PRIVATE /fsanitize=address)
  endif()
  add_test(NAME mef_${src} COMMAND mef_${src})
endfunction()

add_executable(fabric_smoke fabric_smoke.cpp)
target_link_libraries(fabric_smoke PRIVATE mef_core)
mef_apply_warnings(fabric_smoke)
add_test(NAME fabric_smoke COMMAND fabric_smoke)

mef_add_test(core_tests)
mef_add_test(property_tests)
mef_add_test(concurrency_tests)
mef_add_test(adversarial_tests)
mef_add_test(system_tests)
